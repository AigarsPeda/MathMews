const ignore = () => {};

/** Show the alternate question's givens without the original answer controls. */
export function PuzzleResponseModel({ puzzle }: { puzzle: Puzzle }) {
  if (puzzle.type === 'multiple_choice' || !puzzle.type) return <>
    {puzzle.everyday ? <EverydayDiagram model={puzzle.everyday.model} /> : null}
    {puzzle.taskVisual ? <QuantityTaskScene model={puzzle.taskVisual} disabled showExpression={puzzle.difficulty !== 'hard'} /> : null}
  </>;
  if (isBridgePuzzle(puzzle)) {
    const model = puzzle.type === 'coordinate_map' && puzzle.payload.mode === 'plot'
      ? { ...puzzle, payload: { ...puzzle.payload, mode: 'translate' as const, start: puzzle.payload.end, end: { x: puzzle.payload.end.x === puzzle.payload.max ? -1 : 1, y: 0 } } } : puzzle;
    return <BridgeTask puzzle={model} value={null} onChange={ignore} answered readOnly />;
  }
  if (puzzle.type === 'geometry_measure') return <GeometryMeasureTask puzzle={puzzle} value={null} onChange={ignore} answered readOnly />;
  if (puzzle.type === 'number_line') return <>
    <Text style={styles.expression}>{puzzle.payload.start} {puzzle.payload.jump < 0 ? '−' : '+'} {Math.abs(puzzle.payload.jump)} = ?</Text>
    <NumberLineTrack min={puzzle.payload.min} max={puzzle.payload.max} start={puzzle.payload.start} value={puzzle.payload.start} disabled />
  </>;
  if (puzzle.type === 'balance') return <BalanceScale left={<Text style={styles.expression}>{puzzle.payload.leftDisplay}</Text>} right={<Text style={styles.expression}>{puzzle.payload.rightValue}</Text>} />;
  if (puzzle.type === 'true_false') return <Text style={styles.expression}>{puzzle.payload.statement.split('=')[0].trim()} = ?</Text>;
  if (puzzle.type === 'fix_mistake') return <View style={styles.facts}>{puzzle.payload.wrongWork.map((s, i) => <Text key={i} style={styles.expression}>{s}</Text>)}</View>;
  if (puzzle.type === 'pair_sum') return <View style={styles.tiles}>{puzzle.payload.numbers.map((n, i) => <Text key={i} style={styles.tile}>{n}</Text>)}</View>;
  if (puzzle.type === 'fraction_build') return <View style={styles.picture}><FractionPieChart denominator={puzzle.payload.denominator} shaded={0} size={156} /></View>;
  if (puzzle.type === 'fraction_equivalent') return <View style={styles.picture}>
    <Text style={styles.expression}>{puzzle.payload.numerator}/{puzzle.payload.denominator}</Text>
    <FractionPieChart denominator={puzzle.payload.denominator} shaded={puzzle.payload.numerator} size={156} />
  </View>;
  if (puzzle.type === 'fraction_match') {
    const p = puzzle.payload.pairs[0];
    return <View style={styles.picture}><FractionPieChart denominator={p.denominator} shaded={p.numerator} size={156} /></View>;
  }
  if (puzzle.type === 'pattern_next') return <Text style={styles.expression}>{puzzle.payload.sequence.join(', ')}, ?</Text>;
  if (puzzle.type === 'function_machine') return <Text style={styles.expression}>{puzzle.payload.input} → ? → {puzzle.payload.output}</Text>;
  if (puzzle.type === 'estimate') return <Text style={styles.expression}>{puzzle.payload.expression} = ?</Text>;
  if (puzzle.type === 'compare') return <Text style={styles.expression}>A: {puzzle.payload.optionA}{'\n'}B: {puzzle.payload.optionB}</Text>;
  if (puzzle.type === 'fair_share') return <Text style={styles.expression}>{puzzle.payload.items} ÷ {puzzle.payload.people} = ?</Text>;
  if (puzzle.type === 'place_value') {
    const { start, mode, operand = 0 } = puzzle.payload;
    const state = mode === 'build' ? { tens: Math.floor(start / 10), ones: start % 10, added: false } : initialPlaceValue(puzzle);
    return <View style={styles.facts}>
      <Text style={styles.expression}>{start}{mode === 'add' || mode === 'subtract' ? ` ${mode === 'add' ? '+' : '−'} ${operand} = ?` : ''}</Text>
      <VisualPracticeBoard compact tokens={getPlaceValueTokens(state)} boardHeight={140} durationMs={0} staggerMs={0} />
    </View>;
  }
  if (puzzle.type === 'decimal_place_value') {
    const { hundredths, mode } = puzzle.payload;
    const state = mode === 'build' ? { ones: Math.floor(hundredths / 100), tenths: Math.floor(hundredths % 100 / 10), hundredths: hundredths % 10 } : initialDecimalPlace(puzzle);
    return <View style={styles.tiles}>{Object.entries(state).map(([unit, count]) => <View key={unit} style={styles.unit}>
      <Text style={styles.unitLabel}>{unit === 'ones' ? '1' : unit === 'tenths' ? '1/10' : '1/100'}</Text><Text style={styles.expression}>{count}</Text>
    </View>)}</View>;
  }
  return null;
}

type Props = { puzzle: Puzzle; value: ResponseValue; onChange: (value: ResponseValue) => void; answered: boolean; incorrectAttempt?: boolean; isCorrect: boolean };
export function PuzzleResponseTask({ puzzle, value, onChange, answered, incorrectAttempt = false, isCorrect }: Props) {
  const { t, i18n } = useTranslation();
  const response = puzzle.response;
  if (!response) return null;
  const decimalKey = response.kind === 'write' && (response.answer.includes('.') || 'everyday' in puzzle && puzzle.everyday?.model.kind === 'money' && puzzle.everyday.model.unit === 'euros');
  const written = typeof value === 'string' && value ? value.replace('.', /^(lv|ru)/.test(i18n.language) ? ',' : '.') : response.kind === 'write' && response.format === 'clock' ? 'hh:mm' : '?';
  const disabled = answered;
  const change = (next: ResponseValue) => { if (!disabled) onChange(next); };
  return <View style={styles.wrap} testID="puzzle-response">
    <PuzzleResponseModel puzzle={puzzle} />
    {response.kind === 'choose' ? <MultipleChoiceTask puzzle={{ ...puzzle, type: 'multiple_choice', choices: response.options, correctIndex: response.correctIndex, taskVisual: undefined, ...('everyday' in puzzle ? { everyday: undefined } : {}) }} selectedIndex={typeof value === 'string' && value !== '' ? Number(value) : null} onSelect={index => change(String(index))} answered={answered} incorrectAttempt={incorrectAttempt} isCorrect={isCorrect} />
      : response.kind === 'judge' ? <View style={styles.wrap}>
        <Text style={styles.instruction}>{t('activityMix.judge')}</Text>
        <TrueFalseTask puzzle={{ ...puzzle, type: 'true_false', payload: { statement: response.proposed }, isTrue: response.isCorrect }} selectedIndex={value === null ? null : value === true ? 0 : 1} onSelect={index => change(index === 0)} answered={answered} incorrectAttempt={incorrectAttempt} isCorrect={isCorrect} />
      </View> : <View style={styles.entry}>
        <Text style={styles.instruction}>{t(response.format === 'clock' ? 'activityMix.writeTime' : response.format === 'fraction' ? 'activityMix.writeFraction' : 'activityMix.write')}</Text>
        <Text testID="response-value" accessibilityLabel={`${t('activityMix.answer')}: ${written}`} accessibilityLiveRegion="polite" style={[styles.value, incorrectAttempt && styles.wrong, answered && isCorrect && styles.correct]}>{written}</Text>
        {[['1', '2', '3'], ['4', '5', '6'], ['7', '8', '9'], [response.format === 'clock' ? ':' : response.format === 'fraction' ? '/' : decimalKey ? /^(lv|ru)/.test(i18n.language) ? ',' : '.' : '', '0', 'erase']].map((row, i) => <View key={i} style={styles.row}>{row.map(key => key === '' ? <View key="space" style={styles.keySpace} /> : <Pressable key={key} testID={`response-key-${key === ',' ? '.' : key}`} accessibilityRole="button" accessibilityLabel={key === 'erase' ? t('activityMix.erase') : key} accessibilityState={{ disabled }} disabled={disabled} onPress={() => change(editResponse(value, key, response.format))} style={({ pressed }) => [styles.key, disabled && styles.disabled, pressed && styles.pressed]}>
          <Text style={styles.keyText}>{key === 'erase' ? t('geometryModels.eraseShort') : key}</Text>
        </Pressable>)}</View>)}
        <View style={styles.row}>{(response.format === 'number' && response.answer.startsWith('-') ? ['sign', 'clear'] : ['clear']).map(key => <Pressable key={key} testID={`response-key-${key}`} accessibilityRole="button" disabled={disabled} accessibilityState={{ disabled }} accessibilityLabel={t(key === 'sign' ? 'activityMix.sign' : 'activityMix.clear')} onPress={() => change(editResponse(value, key, response.format))} style={styles.textButton}><Text style={styles.textButtonLabel}>{key === 'sign' ? '±' : t('activityMix.clear')}</Text></Pressable>)}</View>
      </View>}
  </View>;
}
const styles = StyleSheet.create({
  wrap: { gap: 12 }, facts: { gap: 8 }, picture: { alignItems: 'center' }, expression: { fontSize: 24, lineHeight: 32, fontWeight: '800', textAlign: 'center', color: C.text },
  instruction: { fontSize: 16, lineHeight: 23, fontWeight: '700', textAlign: 'center', color: C.textMuted }, entry: { width: '100%', maxWidth: 340, alignSelf: 'center', gap: 8 },
  value: { minHeight: 52, padding: 8, borderWidth: 2, borderColor: C.cardBorder, borderRadius: 12, fontSize: 28, fontWeight: '800', textAlign: 'center', color: C.text, fontVariant: ['tabular-nums'] },
  row: { flexDirection: 'row', gap: 8 }, keySpace: { flex: 1 }, key: { flex: 1, minHeight: 48, paddingVertical: 8, paddingHorizontal: 4, borderWidth: 2, borderBottomWidth: 4, borderRadius: 12, borderColor: C.cardBorder, backgroundColor: C.card, alignItems: 'center', justifyContent: 'center' },
  keyText: { fontSize: 22, lineHeight: 28, fontWeight: '800', color: C.text, textAlign: 'center' }, textButton: { flex: 1, minHeight: 44, justifyContent: 'center', alignItems: 'center' }, textButtonLabel: { fontSize: 16, lineHeight: 23, fontWeight: '700', color: C.primaryDark, textAlign: 'center' },
  tiles: { flexDirection: 'row', flexWrap: 'wrap', justifyContent: 'center', gap: 10 }, tile: { minWidth: 54, padding: 12, fontSize: 22, fontWeight: '700', color: C.text, backgroundColor: C.progressTrack, borderRadius: 12, textAlign: 'center' },
  unit: { flex: 1, alignItems: 'center', gap: 8 }, unitLabel: { fontSize: 16, lineHeight: 23, fontWeight: '700', color: C.textMuted }, wrong: { borderColor: C.incorrect }, correct: { borderColor: C.success }, disabled: { opacity: 0.5 }, pressed: { backgroundColor: C.selectedSurface, borderColor: C.primary },
});
