/** Temporary optimized-build A/B fixture. Removed after native performance review. */
import PlayScreen from '@/app/play';
import BeforePlayScreen from '@/scripts/native/puzzle-motion/qa-before-play';
import { PuzzleProgress } from '@/components/puzzle/PuzzleProgress';
import { PuzzleProgress as BeforeProgress } from '@/scripts/native/puzzle-motion/qa-before-progress';
import { GameContext, useGame } from '@/contexts/GameProvider';
import { LocaleContext } from '@/contexts/LocaleProvider';
import { getTopicPaths } from '@/constants/puzzle-path';
import { createDefaultGameSave } from '@/utils/game-storage';
import { applyPuzzleAnswer } from '@/utils/game-operations';
import { grantCatPlanRewards } from '@/utils/cat-plan-rewards';
import type { GameSave } from '@/types/save';
import i18n from '@/i18n';
import { useLocalSearchParams, useRouter } from 'expo-router';
import { useCallback, useEffect, useLayoutEffect, useRef, useState } from 'react';
import { I18nextProvider } from 'react-i18next';
import { Pressable, Text, View } from 'react-native';
import { runOnJS, runOnUI, useFrameCallback, useSharedValue, type FrameCallback } from 'react-native-reanimated';

const [lesson] = getTopicPaths('easy', 'cat-plans');
function replaySave() {
  const save = createDefaultGameSave();
  save.progress.completedPuzzleIds = lesson.levels.flatMap(level => level.puzzleIds);
  return grantCatPlanRewards(save);
}
function useMotionTrace(onReport: (name: string, frames: number[]) => void) {
  const active = useSharedValue(false), start = useSharedValue(0), duration = useSharedValue(500);
  const name = useSharedValue(''), samples = useSharedValue<number[]>([]);
  const callbackRef = useRef<FrameCallback | null>(null);
  const report = useCallback((label: string, values: number[]) => {
    callbackRef.current?.setActive(false);
    onReport(label, values);
  }, [onReport]);
  const frame = useFrameCallback(info => {
    'worklet';
    if (!active.get()) return;
    if (!start.get()) start.set(info.timestamp);
    if (info.timeSincePreviousFrame !== null) samples.modify(values => { values.push(info.timeSincePreviousFrame!); return values; });
    if (info.timestamp - start.get() >= duration.get()) {
      active.set(false);
      runOnJS(report)(name.get(), samples.get());
    }
  }, false);
  callbackRef.current = frame;
  return useCallback((label: string, ms = 500) => {
    runOnUI((label: string, ms: number) => {
      'worklet'; name.set(label); samples.set([]); start.set(0); duration.set(ms); active.set(true);
    })(label, ms);
    frame.setActive(true);
  }, [active, duration, frame, name, samples, start]);
}
export default function MotionFixture() {
  const realGame = useGame(), router = useRouter();
  const params = useLocalSearchParams<{ lesson?: string; level?: string }>();
  const [save, setSave] = useState(replaySave), current = useRef(save);
  const [revision, setRevision] = useState(0), [variant, setVariant] = useState<'before' | 'after'>('before');
  const [probe, setProbe] = useState(false), [step, setStep] = useState(0);
  const [reports, setReports] = useState<{label: string; frames: number[]}[]>([]);
  const onReport = useCallback((label: string, frames: number[]) => setReports(current => [...current, { label, frames }]), []);
  const trace = useMotionTrace(onReport);
  const [language] = useState(() => i18n.cloneInstance({ lng: 'en' }));
  const commit = useCallback((next: GameSave) => { current.current = next; setSave(next); }, []);
  useEffect(() => { router.setParams({ difficulty: 'easy', lesson: lesson.id, level: lesson.levels[0].id, replay: '1' }); }, [router]);
  useLayoutEffect(() => {
    if (!probe && params.level) trace(`${variant}-entry-${params.level}`, 400);
  }, [params.level, probe, variant]);
  useLayoutEffect(() => {
    if (!probe) trace(`${variant}-answer-${save.wallet.coins}`, 650);
  }, [save.wallet.coins]);
  useLayoutEffect(() => { if (probe) trace(`${variant}-bar-${step}`, 500); }, [probe, step, variant]);
  const game = { ...realGame, pet: save.pet, wallet: save.wallet, progress: save.progress, isReady: true, hasCompletedOnboarding: true,
    recordInteraction() { return 0; }, recordLearningHelp() {}, purchaseVisualHelp: () => true,
    answerPuzzle(puzzle: Parameters<typeof realGame.answerPuzzle>[0], correct: boolean, id: string) {
      const result = applyPuzzleAnswer(current.current, puzzle, correct, id);
      if (!result.accepted) return null;
      commit(result.save); return result.coins;
    },
  };
  const Screen = variant === 'before' ? BeforePlayScreen : PlayScreen;
  const Progress = variant === 'before' ? BeforeProgress : PuzzleProgress;
  const resetPlay = () => {
    commit(replaySave());
    router.setParams({ difficulty: 'easy', lesson: lesson.id, level: lesson.levels[0].id, replay: '1' });
    setRevision(value => value + 1);
  };
  return <I18nextProvider i18n={language}><LocaleContext.Provider value={{ locale: 'en', isReady: true, setLocale: async () => {} }}><GameContext.Provider value={game}>
    <View style={{ flex: 1, backgroundColor: '#FFFFFF' }}>
      <View style={{ flex: 1 }}>{probe ? <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center' }}>
        <View style={{ width: 220 }}><Progress steps={Array.from({ length: 10 }, (_, i) => i < step)} label={`Probe ${step}`} /></View>
        <Pressable accessibilityLabel="QA Advance" onPress={() => setStep(value => (value + 1) % 10)} style={{ padding: 24 }}><Text>Advance</Text></Pressable>
      </View> : params.lesson ? <Screen key={revision} /> : null}</View>
      <Text testID="qa-variant">{variant}</Text>
      <Text testID="qa-trace" accessibilityLabel={JSON.stringify(reports)} style={{ height: 18 }}>QA traces {reports.length}</Text>
      <View style={{ flexDirection: 'row', paddingBottom: 24 }}>
        {['Before', 'After', 'Bar', 'Play'].map(label => <Pressable key={label} accessibilityLabel={`QA ${label}`} onPress={() => {
          setReports([]);
          if (label === 'Before' || label === 'After') { setVariant(label === 'Before' ? 'before' : 'after'); setStep(0); resetPlay(); }
          else { setProbe(label === 'Bar'); setStep(0); if (label === 'Play') resetPlay(); }
        }} style={{ padding: 12 }}><Text>{label}</Text></Pressable>)}
      </View>
    </View>
  </GameContext.Provider></LocaleContext.Provider></I18nextProvider>;
}
